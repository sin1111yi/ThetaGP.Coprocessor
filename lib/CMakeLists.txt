# Every library this repository builds against and does not carry: what it is,
# where it comes from, and the one revision of it the code was written for.
# Both builds read this file -- the firmware before project(), the simulator
# after -- so it is include()d rather than added as a subdirectory:
#
#   include(<repo>/lib/CMakeLists.txt)
#   thetagp_require(lvgl nanopb ...)
#
# thetagp_require() puts each library it is given under lib/<name> unless it
# is already there, and leaves the path in THETAGP_<NAME>_DIR. Nothing else
# fetches anything: the ESP-IDF component manager is switched off in the root
# CMakeLists.txt, and the ESP components the firmware uses are pinned here
# like the rest.
#
# ThetaGP.PB, the schema of the link, also lives in lib/ but is a git
# submodule: the repository pins it, not this file.

include_guard(GLOBAL)

set(THETAGP_LIB_DIR "${CMAKE_CURRENT_LIST_DIR}")

# --- How a library gets there -----------------------------------------------

# A git checkout of <repository> at <commit> in <dir>. <marker> is a file the
# source tree is known by.
#
# A source tree that is already there without a .git is used as it is: this
# is how an archive unpacked by hand is honoured, and it needs no network. A
# checkout on the pinned commit is used as it is, and one on another commit
# is moved to it, so changing a pin is enough.
function(_thetagp_git name dir repository commit marker)
  if(EXISTS "${dir}/${marker}" AND NOT EXISTS "${dir}/.git")
    message(STATUS "${name}: using the source tree in ${dir}")
    return()
  endif()

  find_package(Git REQUIRED)

  if(EXISTS "${dir}/.git")
    execute_process(
      COMMAND ${GIT_EXECUTABLE} -C "${dir}" rev-parse HEAD
      OUTPUT_VARIABLE current
      OUTPUT_STRIP_TRAILING_WHITESPACE
      ERROR_QUIET
      RESULT_VARIABLE rev_result)
    if(rev_result EQUAL 0 AND current STREQUAL commit)
      message(STATUS "${name}: ${dir} is at ${commit}")
      return()
    endif()
    message(STATUS "${name}: ${dir} is elsewhere, moving to ${commit}")
  else()
    file(REMOVE_RECURSE "${dir}")
    file(MAKE_DIRECTORY "${dir}")
    execute_process(
      COMMAND ${GIT_EXECUTABLE} init --quiet "${dir}"
      RESULT_VARIABLE init_result
      ERROR_VARIABLE init_error)
    if(NOT init_result EQUAL 0)
      message(FATAL_ERROR "${name}: cannot create ${dir}: ${init_error}")
    endif()
    execute_process(
      COMMAND ${GIT_EXECUTABLE} -C "${dir}" remote add origin "${repository}"
      ERROR_QUIET)
  endif()

  message(STATUS "${name}: fetching ${commit} into ${dir}")

  # One commit is all that is wanted. A server that refuses to hand out a
  # single commit is asked for the history instead, which still lands on the
  # pinned one.
  execute_process(
    COMMAND ${GIT_EXECUTABLE} -C "${dir}" fetch --quiet --depth 1 origin
            "${commit}"
    RESULT_VARIABLE fetch_result
    ERROR_VARIABLE fetch_error)
  if(NOT fetch_result EQUAL 0)
    execute_process(
      COMMAND ${GIT_EXECUTABLE} -C "${dir}" fetch --quiet origin
      RESULT_VARIABLE fetch_result
      ERROR_VARIABLE fetch_error)
  endif()
  if(NOT fetch_result EQUAL 0)
    file(REMOVE_RECURSE "${dir}")
    message(FATAL_ERROR
        "${name}: fetching ${commit} from ${repository} failed: "
        "${fetch_error}")
  endif()

  execute_process(
    COMMAND ${GIT_EXECUTABLE} -C "${dir}" checkout --quiet "${commit}"
    RESULT_VARIABLE checkout_result
    ERROR_VARIABLE checkout_error)
  if(NOT checkout_result EQUAL 0)
    file(REMOVE_RECURSE "${dir}")
    message(FATAL_ERROR
        "${name}: checking out ${commit} failed: ${checkout_error}")
  endif()

  if(NOT EXISTS "${dir}/${marker}")
    file(REMOVE_RECURSE "${dir}")
    message(FATAL_ERROR "${name}: ${dir} came back without a ${marker}")
  endif()
endfunction()

# The archive at <url>, whose SHA-256 is <sha256>, unpacked into <dir>.
#
# The digest of what was unpacked is left beside it in .thetagp-pin: a
# directory that carries the pinned digest is used as it is, one that carries
# another is replaced, and one that carries none was put there by hand and is
# left alone.
function(_thetagp_archive name dir url sha256 marker)
  set(stamp "${dir}/.thetagp-pin")
  if(EXISTS "${dir}/${marker}")
    if(NOT EXISTS "${stamp}")
      message(STATUS "${name}: using the source tree in ${dir}")
      return()
    endif()
    file(READ "${stamp}" current)
    string(STRIP "${current}" current)
    if(current STREQUAL sha256)
      message(STATUS "${name}: ${dir} is the pinned archive")
      return()
    endif()
    message(STATUS "${name}: ${dir} is another archive, replacing it")
  endif()

  set(archive "${CMAKE_BINARY_DIR}/thetagp-downloads/${name}.zip")
  message(STATUS "${name}: downloading ${url}")
  file(DOWNLOAD "${url}" "${archive}"
       EXPECTED_HASH SHA256=${sha256}
       TLS_VERIFY ON
       STATUS status)
  list(GET status 0 status_code)
  if(NOT status_code EQUAL 0)
    list(GET status 1 status_text)
    file(REMOVE "${archive}")
    message(FATAL_ERROR
        "${name}: downloading ${url} failed, or it is not the archive "
        "pinned here (SHA-256 ${sha256}): ${status_text}")
  endif()

  file(REMOVE_RECURSE "${dir}")
  file(MAKE_DIRECTORY "${dir}")
  file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${dir}")
  if(NOT EXISTS "${dir}/${marker}")
    file(REMOVE_RECURSE "${dir}")
    message(FATAL_ERROR "${name}: ${url} unpacked without a ${marker}")
  endif()
  file(WRITE "${stamp}" "${sha256}\n")
endfunction()

# --- What the builds call ---------------------------------------------------

# thetagp_require(<library>...): makes each named library present under lib/
# and sets THETAGP_<NAME>_DIR to where it is. The names are the ones below.
function(thetagp_require)
  foreach(library IN LISTS ARGN)
    if(library STREQUAL "lvgl")
      set(dir "${THETAGP_LIB_DIR}/lvgl")
      _thetagp_git("LVGL" "${dir}" "${THETAGP_LVGL_REPOSITORY}"
                   "${THETAGP_LVGL_TAG}" "CMakeLists.txt")
      set(THETAGP_LVGL_DIR "${dir}" PARENT_SCOPE)
    elseif(library STREQUAL "nanopb")
      set(dir "${THETAGP_LIB_DIR}/nanopb")
      _thetagp_git("nanopb" "${dir}" "${THETAGP_NANOPB_REPOSITORY}"
                   "${THETAGP_NANOPB_TAG}" "pb.h")
      set(THETAGP_NANOPB_DIR "${dir}" PARENT_SCOPE)
    elseif(library STREQUAL "sdl2")
      set(dir "${THETAGP_LIB_DIR}/sdl2")
      _thetagp_git("SDL2" "${dir}" "${THETAGP_SDL2_REPOSITORY}"
                   "${THETAGP_SDL2_TAG}" "include/SDL.h")
      set(THETAGP_SDL2_DIR "${dir}" PARENT_SCOPE)
    elseif(library STREQUAL "esp_lvgl_adapter")
      set(dir "${THETAGP_LIB_DIR}/espressif__esp_lvgl_adapter")
      set(version "${THETAGP_ESP_LVGL_ADAPTER_VERSION}")
      _thetagp_archive("esp_lvgl_adapter" "${dir}"
          "${THETAGP_ESP_COMPONENTS_URL}/esp_lvgl_adapter/${version}/espressif__esp_lvgl_adapter-v${version}.zip"
          "${THETAGP_ESP_LVGL_ADAPTER_SHA256}" "CMakeLists.txt")
      set(THETAGP_ESP_LVGL_ADAPTER_DIR "${dir}" PARENT_SCOPE)
    elseif(library STREQUAL "esp_lcd_touch")
      set(dir "${THETAGP_LIB_DIR}/espressif__esp_lcd_touch")
      set(version "${THETAGP_ESP_LCD_TOUCH_VERSION}")
      _thetagp_archive("esp_lcd_touch" "${dir}"
          "${THETAGP_ESP_COMPONENTS_URL}/esp_lcd_touch/${version}/espressif__esp_lcd_touch-v${version}.zip"
          "${THETAGP_ESP_LCD_TOUCH_SHA256}" "CMakeLists.txt")
      set(THETAGP_ESP_LCD_TOUCH_DIR "${dir}" PARENT_SCOPE)
    else()
      message(FATAL_ERROR
          "thetagp_require: ${library} is not a library lib/CMakeLists.txt "
          "knows; the names are lvgl, nanopb, sdl2, esp_lvgl_adapter and "
          "esp_lcd_touch.")
    endif()
  endforeach()
endfunction()
